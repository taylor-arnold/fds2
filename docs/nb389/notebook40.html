    anno

word_counts = (
    anno
    .filter(c.is_alpha)
    .group_by(c.lemma)
    .agg(word_count = pl.len())
)

total_words = word_counts.select(c.word_count.sum()).item()
total_bigrams = bigrams.height

pmi = (
    bigrams
    .group_by(c.bigram, c.lemma, c.next_token)
    .agg(bigram_count = pl.len())
    .join(
        word_counts.rename({"lemma": "w1", "word_count": "w1_count"}),
        left_on=c.lemma,
        right_on=c.w1
    )
    .join(
        word_counts.rename({"lemma": "w2", "word_count": "w2_count"}),
        left_on=c.next_token,
        right_on=c.w2
    )
    .with_columns(
        p_bigram = c.bigram_count / total_bigrams,
        p_w1 = c.w1_count / total_words,
        p_w2 = c.w2_count / total_words
    )
    .with_columns(
        pmi = (c.p_bigram / (c.p_w1 * c.p_w2)).log()
    )
    .select(c.bigram, c.bigram_count, c.pmi)
)

7. The block below is given. It follows the chapter's steps to compute the PMI of
every bigram and saves the result as `pmi`, with one row per bigram. Run it and look
at the table.

```{python}
#| tags: [noclear]
bigrams = (
    anno
    .filter(c.is_alpha)
    .with_columns(
        next_token = c.lemma.shift(-1).over([c.doc_id, c.sid])
    )
    .filter(c.next_token.is_not_null())
    .with_columns(
        bigram = c.lemma + " " + c.next_token
    )
)

word_counts = (
    anno
    .filter(c.is_alpha)
    .group_by(c.lemma)
    .agg(word_count = pl.len())
)

total_words = word_counts.select(c.word_count.sum()).item()
total_bigrams = bigrams.height

pmi = (
    bigrams
    .group_by(c.bigram, c.lemma, c.next_token)
    .agg(bigram_count = pl.len())
    .join(
        word_counts.rename({"lemma": "w1", "word_count": "w1_count"}),
        left_on=c.lemma,
        right_on=c.w1
    )
    .join(
        word_counts.rename({"lemma": "w2", "word_count": "w2_count"}),
        left_on=c.next_token,
        right_on=c.w2
    )
    .with_columns(
        p_bigram = c.bigram_count / total_bigrams,
        p_w1 = c.w1_count / total_words,
        p_w2 = c.w2_count / total_words
    )
    .with_columns(
        pmi = (c.p_bigram / (c.p_w1 * c.p_w2)).log()
    )
    .select(c.bigram, c.bigram_count, c.pmi)
)
pmi
```
